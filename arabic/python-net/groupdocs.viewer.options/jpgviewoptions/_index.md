### مثال

```python
from groupdocs.viewer import Viewer
from groupdocs.viewer.options import JpgViewOptions

def render_to_jpg():
    # تحميل مستند (مثال: Excel أو Numbers أو صورة)
    with Viewer("invoice.xlsx") as viewer:
        # إنشاء خيارات عرض JPG باستخدام نمط ملف الإخراج.
        # {0} سيتم استبداله برقم الصفحة.
        view_options = JpgViewOptions("output_{0}.jpg")
        # تعيين حجم الصورة الناتجة المطلوب بالبكسل.
        view_options.width = 800
        view_options.height = 900
        # تحويل المستند إلى ملفات JPEG.
        viewer.view(view_options)

if __name__ == "__main__":
    render_to_jpg()
```
