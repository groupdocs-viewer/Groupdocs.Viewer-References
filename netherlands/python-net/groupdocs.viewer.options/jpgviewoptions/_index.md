### Voorbeeld

```python
from groupdocs.viewer import Viewer
from groupdocs.viewer.options import JpgViewOptions

def render_to_jpg():
    # Laad een document (bijv. Excel, Numbers of afbeelding)
    with Viewer("invoice.xlsx") as viewer:
        # Maak JPG-weergave-opties met een uitvoerbestands-patroon.
        # {0} wordt vervangen door het paginanummer.
        view_options = JpgViewOptions("output_{0}.jpg")
        # Stel de gewenste uitvoerafbeeldingsgrootte in pixels in.
        view_options.width = 800
        view_options.height = 900
        # Render het document naar JPEG‑bestanden.
        viewer.view(view_options)

if __name__ == "__main__":
    render_to_jpg()
```
