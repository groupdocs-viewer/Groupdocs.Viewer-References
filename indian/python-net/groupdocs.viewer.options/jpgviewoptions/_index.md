### उदाहरण

```python
from groupdocs.viewer import Viewer
from groupdocs.viewer.options import JpgViewOptions

def render_to_jpg():
    # एक दस्तावेज़ लोड करें (उदा., Excel, Numbers, या छवि)
    with Viewer("invoice.xlsx") as viewer:
        # आउटपुट फ़ाइल पैटर्न के साथ JPG व्यू विकल्प बनाएं।
        # {0} को पृष्ठ संख्या से बदल दिया जाएगा।
        view_options = JpgViewOptions("output_{0}.jpg")
        # वांछित आउटपुट छवि आकार को पिक्सेल में सेट करें।
        view_options.width = 800
        view_options.height = 900
        # दस्तावेज़ को JPEG फ़ाइलों में रेंडर करें।
        viewer.view(view_options)

if __name__ == "__main__":
    render_to_jpg()
```
