### Exempel

```python
from groupdocs.viewer import Viewer
from groupdocs.viewer.options import JpgViewOptions

def render_to_jpg():
    # Läs in ett dokument (t.ex. Excel, Numbers eller bild)
    with Viewer("invoice.xlsx") as viewer:
        # Skapa JPG-visningsalternativ med ett utdatafilsmönster.
        # {0} kommer att ersättas med sidnumret.
        view_options = JpgViewOptions("output_{0}.jpg")
        # Ställ in önskad storlek på utdata bilden i pixlar.
        view_options.width = 800
        view_options.height = 900
        # Rendera dokumentet till JPEG-filer.
        viewer.view(view_options)

if __name__ == "__main__":
    render_to_jpg()
```
