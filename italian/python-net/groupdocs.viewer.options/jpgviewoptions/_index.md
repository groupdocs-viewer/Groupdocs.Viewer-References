### Esempio

```python
from groupdocs.viewer import Viewer
from groupdocs.viewer.options import JpgViewOptions

def render_to_jpg():
    # Carica un documento (ad es., Excel, Numbers o immagine)
    with Viewer("invoice.xlsx") as viewer:
        # Crea opzioni di visualizzazione JPG con un modello di file di output.
        # {0} verrà sostituito dal numero di pagina.
        view_options = JpgViewOptions("output_{0}.jpg")
        # Imposta la dimensione desiderata dell'immagine di output in pixel.
        view_options.width = 800
        view_options.height = 900
        # Esegui il rendering del documento in file JPEG.
        viewer.view(view_options)

if __name__ == "__main__":
    render_to_jpg()
```
