### Örnek

```python
from groupdocs.viewer import Viewer
from groupdocs.viewer.options import JpgViewOptions

def render_to_jpg():
    # Bir belge yükle (ör. Excel, Numbers veya görüntü)
    with Viewer("invoice.xlsx") as viewer:
        # Çıktı dosya deseniyle JPG görüntü seçenekleri oluştur.
        # {0} sayfa numarasıyla değiştirilecektir.
        view_options = JpgViewOptions("output_{0}.jpg")
        # İstenen çıktı görüntü boyutunu piksel olarak ayarla.
        view_options.width = 800
        view_options.height = 900
        # Belgeyi JPEG dosyalarına render et.
        viewer.view(view_options)

if __name__ == "__main__":
    render_to_jpg()
```
