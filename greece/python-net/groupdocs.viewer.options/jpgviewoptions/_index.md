### Παράδειγμα

```python
from groupdocs.viewer import Viewer
from groupdocs.viewer.options import JpgViewOptions

def render_to_jpg():
    # Φορτώστε ένα έγγραφο (π.χ., Excel, Numbers ή εικόνα)
    with Viewer("invoice.xlsx") as viewer:
        # Δημιουργήστε επιλογές προβολής JPG με ένα πρότυπο αρχείου εξόδου.
        # {0} θα αντικατασταθεί με τον αριθμό σελίδας.
        view_options = JpgViewOptions("output_{0}.jpg")
        # Ορίστε το επιθυμητό μέγεθος εξαγόμενης εικόνας σε εικονοστοιχεία.
        view_options.width = 800
        view_options.height = 900
        # Αποδώστε το έγγραφο σε αρχεία JPEG.
        viewer.view(view_options)

if __name__ == "__main__":
    render_to_jpg()
```
