### Contoh

```python
from groupdocs.viewer import Viewer
from groupdocs.viewer.options import JpgViewOptions

def render_to_jpg():
    # Muat dokumen (mis., Excel, Numbers, atau gambar)
    with Viewer("invoice.xlsx") as viewer:
        # Buat opsi tampilan JPG dengan pola file output.
        # {0} akan diganti dengan nomor halaman.
        view_options = JpgViewOptions("output_{0}.jpg")
        # Atur ukuran gambar output yang diinginkan dalam piksel.
        view_options.width = 800
        view_options.height = 900
        # Render dokumen ke file JPEG.
        viewer.view(view_options)

if __name__ == "__main__":
    render_to_jpg()
```
